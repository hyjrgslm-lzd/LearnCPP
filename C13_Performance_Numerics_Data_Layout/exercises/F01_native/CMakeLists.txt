cmake_minimum_required(VERSION 3.28)
project(C13_F01_native LANGUAGES CXX)
include(../cmake/StudySetup.cmake)
if(NOT C13_ENABLE_FRONTIER)
  message(STATUS "[C13] native frontier explicitly disabled")
  return()
endif()
unset(CMAKE_CXX_STANDARD)
if(MSVC)
  set(CMAKE_REQUIRED_FLAGS "/std:c++latest /Zc:__cplusplus")
else()
  set(CMAKE_REQUIRED_FLAGS "-std=c++2c")
endif()
include(CheckCXXSourceCompiles)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/capabilities.txt" "Native interfaces; failed probes are unavailable, not passed examples.\n")
function(c13_native name code)
  string(TOUPPER "${name}" upper)
  check_cxx_source_compiles("${code}" C13_NATIVE_${upper})
  if(C13_NATIVE_${upper})
    add_executable(c13_native_${name} ${name}.cpp)
    target_include_directories(c13_native_${name} PRIVATE "${C13_EXERCISES}/include"
      "${C13_REPO}/C01_Build_Compile_Link/exercises/include")
    if(MSVC)
      target_compile_options(c13_native_${name} PRIVATE /std:c++latest /Zc:__cplusplus /utf-8 /EHsc /W4 /permissive- /fp:precise)
    else()
      target_compile_options(c13_native_${name} PRIVATE -std=c++2c -Wall -Wextra -Wpedantic)
    endif()
    c13_test(c13_native_${name} LABELS native)
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/capabilities.txt" "${name}: available; see subject build and runtime checks\n")
  else()
    message(STATUS "[C13] native ${name}: unavailable; source retained, subject not validated")
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/capabilities.txt" "${name}: unavailable; subject not validated\n")
  endif()
endfunction()
c13_native(simd "#include <simd>\nint main(){std::simd::vec<float,4> x(1.0f); return std::simd::reduce(x)==4?0:1;}")
c13_native(linalg "#include <linalg>\n#include <mdspan>\nint main(){double x[2]={1,2}; std::mdspan<double,std::extents<size_t,2>> a(x); return std::linalg::dot(a,a,0.0)==5?0:1;}")
c13_native(submdspan "#include <mdspan>\nint main(){double x[4]{}; std::mdspan<double,std::extents<size_t,2,2>> a(x); auto s=std::submdspan(a,0,std::full_extent); return int(s.extent(0));}")
c13_native(padded "#include <mdspan>\nint main(){double x[8]{}; std::mdspan<double,std::extents<size_t,2,3>,std::layout_right_padded<4>> a(x); return int(a.mapping().stride(0));}")
c13_native(aligned "#include <mdspan>\nint main(){alignas(64) double x[4]{}; std::mdspan<double,std::extents<size_t,4>,std::layout_right,std::aligned_accessor<double,64>> a(x); return int(a[0]);}")
c13_native(copy_fill "#include <mdspan>\nint main(){double x[4]{},y[4]{}; std::mdspan<double,std::extents<size_t,2,2>> a(x),b(y); std::fill(a,1.0); std::copy(a,b); return int(y[0]);}")
