cmake_minimum_required(VERSION 3.28)
project(c09_review_mini_task_sender LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/Zc:__cplusplus /utf-8 /Zc:preprocessor /FS)
endif()

set(STDEXEC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(STDEXEC_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(STDEXEC_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(STDEXEC_BUILD_RELACY_TESTS OFF CACHE BOOL "" FORCE)
set(STDEXEC_BUILD_PARALLEL_SCHEDULER OFF CACHE BOOL "" FORCE)
set(STDEXEC_ENABLE_ASIO OFF CACHE BOOL "" FORCE)
set(STDEXEC_ENABLE_CUDA OFF CACHE BOOL "" FORCE)
set(STDEXEC_ENABLE_EXTRA_TYPE_CHECKING OFF CACHE BOOL "" FORCE)
set(STDEXEC_ENABLE_IO_URING OFF CACHE BOOL "" FORCE)
set(STDEXEC_ENABLE_NUMA OFF CACHE BOOL "" FORCE)
set(STDEXEC_ENABLE_TASKFLOW OFF CACHE BOOL "" FORCE)
set(STDEXEC_ENABLE_TBB OFF CACHE BOOL "" FORCE)
set(STDEXEC_INSTALL OFF CACHE BOOL "" FORCE)

if(NOT DEFINED STDEXEC_SOURCE_DIR OR NOT EXISTS "${STDEXEC_SOURCE_DIR}/include/stdexec/execution.hpp")
  message(FATAL_ERROR "Set STDEXEC_SOURCE_DIR to cached stdexec source")
endif()
add_subdirectory("${STDEXEC_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/_stdexec-build")
if(NOT TARGET stdexec::stdexec)
  if(TARGET STDEXEC::stdexec)
    add_library(c09_review_stdexec_bridge INTERFACE)
    target_link_libraries(c09_review_stdexec_bridge INTERFACE STDEXEC::stdexec)
    add_library(stdexec::stdexec ALIAS c09_review_stdexec_bridge)
  elseif(TARGET stdexec)
    add_library(stdexec::stdexec ALIAS stdexec)
  else()
    message(FATAL_ERROR "stdexec target missing after add_subdirectory")
  endif()
endif()

set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../../..")
set(MINI_DIR "${ROOT}/C09_Coroutines/exercises/Capstone5_mini_corolib")
set(COMMON_INCLUDE "${ROOT}/C09_Coroutines/exercises/include")
set(TEST_SRC "${MINI_DIR}/tests/task_sender_test.cpp")

function(add_sender_case target overlay_dir)
  add_executable(${target} "${TEST_SRC}")
  target_include_directories(${target} PRIVATE
    "${overlay_dir}"
    "${MINI_DIR}/include"
    "${MINI_DIR}/tests"
    "${COMMON_INCLUDE}")
  target_link_libraries(${target} PRIVATE stdexec::stdexec)
endfunction()

add_sender_case(task_sender_default "${MINI_DIR}/include")
add_sender_case(task_sender_good "${CMAKE_CURRENT_LIST_DIR}/../good_task_sender/include")
add_sender_case(task_sender_bad_wrong_value "${CMAKE_CURRENT_LIST_DIR}/../bad_task_sender_wrong_value/include")
add_sender_case(task_sender_bad_double_complete "${CMAKE_CURRENT_LIST_DIR}/../bad_task_sender_double_complete/include")
