307:class shared_task {
308:public:
309:    explicit shared_task(task<T> source) : state_(std::make_shared<shared_state<T>>()) {
310:        state_->source.emplace(std::move(source));
311:    }
312:
313:    struct awaiter {
314:        using registration = typename shared_state<T>::registration;
315:        using phase = typename registration::phase;
316:        std::shared_ptr<shared_state<T>> state;
317:        std::shared_ptr<registration> node = std::make_shared<registration>();
318:
319:        explicit awaiter(std::shared_ptr<shared_state<T>> s) : state(std::move(s)) {}
320:        awaiter(const awaiter&) = delete;
321:        awaiter(awaiter&&) noexcept = default;
322:        ~awaiter() { if (node) node->status.store(phase::cancelled, std::memory_order_release); }
323:
324:        bool await_ready() const {
325:            std::lock_guard lock{state->mutex};
326:            return state->done;
327:        }
328:        bool await_suspend(std::coroutine_handle<> caller) {
329:            // Locals survive even if completion resumes and destroys this awaiter.
330:            auto shared = state;
331:            auto registration_node = node;
332:            registration_node->caller = caller;
333:            bool start = false;
334:            {
335:                std::lock_guard lock{shared->mutex};
336:                if (shared->done) return false;
337:                shared->waiters.push_back(registration_node);
338:                start = !std::exchange(shared->started, true);
339:            }
340:            if (start) {
341:                try { run(shared); }
342:                catch (...) {
343:                    shared->error = std::current_exception();
344:                    publish(shared);
345:                }
346:            }
347:            auto expected = phase::registering;
348:            return registration_node->status.compare_exchange_strong(
349:                expected, phase::suspended, std::memory_order_acq_rel);
350:        }
351:        T await_resume() const {
352:            if (state->error) std::rethrow_exception(state->error);
353:            return *state->value;
354:        }
355:
356:        // The producer owns its state until source completion. No state/runner cycle.
357:        struct producer {
358:            struct promise_type {
359:                producer get_return_object() noexcept { return {}; }
360:                std::suspend_never initial_suspend() noexcept { return {}; }
361:                std::suspend_never final_suspend() noexcept { return {}; }
362:                void return_void() noexcept {}
363:                void unhandled_exception() noexcept { std::terminate(); }
364:            };
365:        };

368:            try {
369:                state->value.emplace(co_await std::move(*state->source));
370:            } catch (...) {
371:                state->error = std::current_exception();

374:        }
