0002: // Teaching lazy_task<T>.
0003: //
0004: // A lazy_task owns one coroutine frame and can be consumed once. sync_wait
0005: // starts the root coroutine once, then waits for final_suspend to signal
0006: // completion; it does not blindly resume through arbitrary suspension.

0038:     std::condition_variable cv;
0039:     bool done = false;

0042: inline void notify_sync_wait(void* p) noexcept {
0043:     auto& state = *static_cast<sync_wait_state*>(p);
0044:     {
0045:         std::lock_guard lock(state.mutex);
0046:         state.done = true;
0047:     }
0048:     state.cv.notify_one();
0049: }

0051: struct lazy_promise_base {
0052:     std::coroutine_handle<> continuation{};
0053:     void* completion_state{};
0054:     void (*notify_completion)(void*) noexcept {};
0055:     bool started = false;
0056:     std::exception_ptr exception;
0057: 
0058:     static void* operator new(std::size_t size) noexcept {
0059:         lazy_task_last_allocation_size = size;
0060:         if (std::exchange(lazy_task_fail_next_allocation, false)) return nullptr;
0061:         return ::operator new(size, std::nothrow);
0062:     }
0063: 
0064:     static void operator delete(void* p, std::size_t) noexcept {
0065:         ::operator delete(p);
0066:     }
0067: 
0068:     std::suspend_always initial_suspend() noexcept { return {}; }
0069: 
0070:     struct final_awaiter {
0071:         bool await_ready() noexcept { return false; }

0073:         template <class Promise>
0074:         std::coroutine_handle<> await_suspend(std::coroutine_handle<Promise> h) noexcept {
0075:             auto& p = h.promise();
0076:             if (p.continuation) return p.continuation;
0077:             if (p.notify_completion) p.notify_completion(p.completion_state);
0078:             return std::noop_coroutine();
0079:         }

0081:         void await_resume() noexcept {}

0117:             reset();

0290: T sync_wait(lazy_task<T>&& task) {
0291:     lazy_task<T> owned = std::move(task);
0292:     if (!owned.h_) throw std::bad_alloc{};
0293: 
0294:     detail::sync_wait_state state;
0295:     auto& p = owned.h_.promise();
0296:     if (p.started) throw std::logic_error("sync_wait requires an unstarted lazy_task");
0297:     p.completion_state = &state;
0298:     p.notify_completion = detail::notify_sync_wait;
0299:     owned.start();
0300: 
0301:     std::unique_lock lock(state.mutex);
0302:     state.cv.wait(lock, [&] { return state.done; });
0303:     lock.unlock();
0304: 
0305:     if (p.exception) std::rethrow_exception(p.exception);
0306:     if (!p.result) throw std::logic_error("lazy_task completed without a value");
0307:     return std::move(*p.result);
0308: }

0060:         for (;;) {
0061:             std::coroutine_handle<> h;
0062:             {
0063:                 std::unique_lock lock(mutex_);
0064:                 cv_.wait(lock, [&] { return stopped_ || !queue_.empty(); });
0065:                 if (queue_.empty()) return;
0066:                 h = queue_.front();
0067:                 queue_.pop_front();
0068:             }
0069:             if (h && !h.done()) h.resume();
0070:         }
0071:     }

0074:         std::coroutine_handle<> h;
0075:         {
0076:             std::lock_guard lock(mutex_);
0077:             if (queue_.empty()) return false;
0078:             h = queue_.front();
0079:             queue_.pop_front();
0080:         }
0081:         if (h && !h.done()) h.resume();
0082:         return true;
0083:     }

0086:         {
0087:             std::lock_guard lock(mutex_);
0088:             stopped_ = true;
0089:         }

0162:     std::stop_source& stop_source,
0163:     lazy_task<T> first,
0164:     lazy_task<T> second
0165: ) {
0166:     struct shared_state {
0167:         std::mutex mutex;
0168:         std::condition_variable cv;
0169:         std::optional<when_any_result<T>> result;
0170:         std::exception_ptr exception;

0173:     auto run_one = [&](std::size_t index, lazy_task<T> task) mutable {
0174:         try {
0175:             T value = sync_wait(std::move(task));
0176:             {
0177:                 std::lock_guard lock(state.mutex);
0178:                 if (!state.result && !state.exception) {
0179:                     state.result.emplace(when_any_result<T>{index, std::move(value)});
0180:                     stop_source.request_stop();
0181:                 }
0182:             }
0183:             state.cv.notify_one();
0184:         } catch (...) {
0185:             {
0186:                 std::lock_guard lock(state.mutex);
0187:                 if (!state.result && !state.exception) {
0188:                     state.exception = std::current_exception();
0189:                     stop_source.request_stop();
0190:                 }
0191:             }
0192:             state.cv.notify_one();
0193:         }
0194:     };
0195: 
0196:     std::jthread t0(run_one, 0, std::move(first));
0197:     std::jthread t1(run_one, 1, std::move(second));
0198: 
0199:     std::unique_lock lock(state.mutex);
0200:     state.cv.wait(lock, [&] { return state.result || state.exception; });
0201:     if (state.exception) std::rethrow_exception(state.exception);
0202:     return std::move(*state.result);

0035:     bool started{};

0042:     struct final_awaiter {
0043:         bool await_ready() const noexcept { return false; }
0044:         template <typename Promise>
0045:         std::coroutine_handle<> await_suspend(std::coroutine_handle<Promise> h) const noexcept {
0046:             auto& p = h.promise();
0047:             // Completion may let another thread destroy this suspended frame.
0048:             // Read everything needed before publishing completion.
0049:             const auto callback = p.on_completed;
0050:             void* state = p.completion_state;
0051:             const auto continuation = p.continuation;
0052:             if (callback) return callback(state);
0053:             return continuation ? continuation : std::noop_coroutine();
0054:         }
0055:         void await_resume() const noexcept {}
0056:     };
0057: 
0058:     final_awaiter final_suspend() noexcept { return {}; }

0339:                 state->value.emplace(co_await std::move(*state->source));
