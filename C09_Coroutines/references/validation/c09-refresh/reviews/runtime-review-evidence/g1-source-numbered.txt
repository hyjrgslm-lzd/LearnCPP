  31: class shared_task {
  32:     struct control_block;
  33: public:
  34:     struct promise_type {
  35:         std::weak_ptr<control_block> owner;
  36:         shared_task get_return_object();
  37:         std::suspend_always initial_suspend() noexcept { return {}; }
  38:         struct final_awaiter {
  39:             bool await_ready() noexcept { return false; }
  40:             std::coroutine_handle<> await_suspend(std::coroutine_handle<promise_type> h) noexcept {
  41:                 auto cb = h.promise().owner.lock();
  42:                 if (!cb) return std::noop_coroutine();
  43:                 cb->completed = true;
  44:                 auto waiters = std::move(cb->waiters);
  45:                 for (std::size_t i = 1; i < waiters.size(); ++i) waiters[i].resume();
  46:                 return waiters.empty() ? std::noop_coroutine() : waiters[0];
  47:             }
  48:             void await_resume() noexcept {}
  49:         };
  50:         final_awaiter final_suspend() noexcept { return {}; }
  51:         void return_value(T v) noexcept { value.emplace(std::move(v)); }
  52:         void unhandled_exception() noexcept { error = std::current_exception(); }
  53:         std::optional<T> value;
  54:         std::exception_ptr error;
  55:     };

  57:     shared_task() = default;
  58:     explicit shared_task(std::shared_ptr<control_block> cb) : cb_(std::move(cb)) {}
  59:     void resume() const { if (cb_ && cb_->h && !cb_->h.done()) cb_->h.resume(); }
  60:     struct awaiter {
  61:         std::shared_ptr<control_block> cb;
  62:         bool await_ready() const noexcept { return cb->completed; }
  63:         std::coroutine_handle<> await_suspend(std::coroutine_handle<> caller) {
  64:             cb->waiters.push_back(caller);
  65:             if (!std::exchange(cb->started, true)) return cb->h;
  66:             return std::noop_coroutine();

  69:             auto& p = cb->h.promise();
