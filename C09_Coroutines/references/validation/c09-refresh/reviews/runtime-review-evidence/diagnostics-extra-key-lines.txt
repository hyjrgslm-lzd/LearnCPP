12:struct worker_group {
13:    template <class Fn>
14:    void submit(Fn&& fn) {
15:        std::lock_guard lock{mutex};
16:        workers.emplace_back(std::forward<Fn>(fn));
17:    }
18:
19:    void join() {
20:        for (;;) {
21:            std::vector<std::jthread> current;
22:            {
23:                std::lock_guard lock{mutex};
24:                if (workers.empty()) return;
25:                current.swap(workers);
26:            }
27:            for (auto& worker : current) {
28:                if (worker.joinable()) worker.join();
29:            }
30:        }
31:    }
