cmake_minimum_required(VERSION 3.25)
project(H1_as_awaitable LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)
    add_compile_options(/Zc:__cplusplus /utf-8 /Zc:preprocessor)
endif()

if(COMMAND coroutine_study_add_exercise)
    add_executable(H1_as_awaitable checks/main.cpp)
    target_compile_features(H1_as_awaitable PRIVATE cxx_std_23)
    target_include_directories(H1_as_awaitable PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${COROUTINE_STUDY_EXERCISES_DIR}/include")
    target_link_libraries(H1_as_awaitable PRIVATE stdexec::stdexec)
    if(BUILD_TESTING AND COROUTINE_STUDY_TEST_STARTERS)
        add_test(NAME H1_as_awaitable COMMAND H1_as_awaitable)
        coroutine_study_set_test_defaults(H1_as_awaitable starter 30)
    endif()
    coroutine_study_add_checker_variants(H1_as_awaitable)
    if(COROUTINE_STUDY_BUILD_REFERENCE)
        add_executable(H1_as_awaitable_reference solution.cpp)
        target_compile_features(H1_as_awaitable_reference PRIVATE cxx_std_23)
        target_include_directories(H1_as_awaitable_reference PRIVATE "${COROUTINE_STUDY_EXERCISES_DIR}/include")
        target_link_libraries(H1_as_awaitable_reference PRIVATE stdexec::stdexec)
        if(BUILD_TESTING)
            add_test(NAME H1_as_awaitable_reference COMMAND H1_as_awaitable_reference)
            coroutine_study_set_test_defaults(H1_as_awaitable_reference reference 30)
        endif()
    endif()
else()
    add_executable(H1_as_awaitable checks/main.cpp)
    target_include_directories(H1_as_awaitable PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(H1_as_awaitable PRIVATE stdexec::stdexec)
    add_executable(H1_as_awaitable_reference solution.cpp)
    target_link_libraries(H1_as_awaitable_reference PRIVATE stdexec::stdexec)
endif()
