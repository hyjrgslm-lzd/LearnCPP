cmake_minimum_required(VERSION 3.28)
project(h3_doc_alias_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_CXX26_STANDARD_COMPILE_OPTION "/std:c++latest")
  set(CMAKE_CXX26_EXTENSION_COMPILE_OPTION "/std:c++latest")
  add_compile_options(/utf-8 /EHsc /permissive- /Zc:__cplusplus)
endif()
add_executable(h3_doc_alias_probe h3_doc_alias_probe.cpp)
