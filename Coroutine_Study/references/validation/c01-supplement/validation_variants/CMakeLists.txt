cmake_minimum_required(VERSION 3.28)
project(capstone5_validation_variants LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/Zc:__cplusplus /utf-8 /await:strict)
endif()
set(COROUTINE_STUDY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." CACHE PATH "Path to the Coroutine_Study course root")
function(add_variant_test variant test_name source_name)
  add_executable(${variant}_${test_name} "${COROUTINE_STUDY_ROOT}/exercises/Capstone5_mini_corolib/tests/${source_name}")
  target_include_directories(${variant}_${test_name} PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/${variant}/include"
    "${COROUTINE_STUDY_ROOT}/exercises/include")
  add_test(NAME ${variant}_${test_name} COMMAND ${variant}_${test_name})
  set_tests_properties(${variant}_${test_name} PROPERTIES TIMEOUT 30)
endfunction()
add_variant_test(good sync_wait sync_wait_test.cpp)
add_variant_test(good when_all when_all_test.cpp)
add_variant_test(good scope scope_test.cpp)
add_variant_test(good bridge as_awaitable_test.cpp)
add_variant_test(sync_wait_bad_nullopt sync_wait sync_wait_test.cpp)
add_variant_test(sync_wait_bad_error_channel sync_wait sync_wait_test.cpp)
add_variant_test(scope_bad_noop scope scope_test.cpp)
add_variant_test(when_all_bad_error when_all when_all_test.cpp)
add_variant_test(bridge_bad_stopped_value bridge as_awaitable_test.cpp)
