cmake_minimum_required(VERSION 3.28)
project(C11_L09_quic LANGUAGES CXX)
include(../cmake/StudySetup.cmake)
if(C11_ENABLE_QUIC AND C11_BUILD_REFERENCE)
  add_executable(C11_L09_quic main.cpp)
  c11_target(C11_L09_quic)
  target_include_directories(C11_L09_quic PRIVATE "${C11_MSQUIC_SOURCE}/src/inc" "${C11_QUIC_TLS_ROOT}/include")
  target_link_libraries(C11_L09_quic PRIVATE msquic "${C11_QUIC_TLS_ROOT}/lib/libcrypto.lib" crypt32 advapi32 user32 gdi32)
  add_custom_command(TARGET C11_L09_quic POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:msquic>" "$<TARGET_FILE_DIR:C11_L09_quic>" VERBATIM)
  c11_test(C11_L09_quic C11_L09_quic LABEL reference TIMEOUT 40)
else()
  message(STATUS "C11 QUIC: NOT_ENABLED")
endif()
if(TARGET C11_L09_quic)
  learncpp_setup_ide("C11_L09_quic")
else()
  learncpp_setup_ide("")
endif()
