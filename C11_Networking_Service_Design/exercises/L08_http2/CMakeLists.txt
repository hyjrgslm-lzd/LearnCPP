cmake_minimum_required(VERSION 3.28)
project(C11_L08_http2 LANGUAGES CXX)
include(../cmake/StudySetup.cmake)
if(C11_ENABLE_HTTP2 AND C11_BUILD_REFERENCE)
  add_executable(C11_L08_http2 main.cpp)
  c11_target(C11_L08_http2)
  target_link_libraries(C11_L08_http2 PRIVATE nghttp2)
  target_compile_definitions(C11_L08_http2 PRIVATE NGHTTP2_STATICLIB NGHTTP2_NO_SSIZE_T)
  c11_test(C11_L08_http2 C11_L08_http2 LABEL reference)
else()
  message(STATUS "C11 HTTP2: NOT_ENABLED")
endif()
if(TARGET C11_L08_http2)
  learncpp_setup_ide("C11_L08_http2")
else()
  learncpp_setup_ide("")
endif()
