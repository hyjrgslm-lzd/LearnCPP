if(C11_BUILD_REFERENCE)
  add_executable(C11_L04_reactor reactor.cpp)
  c11_target(C11_L04_reactor)
  c11_test(C11_L04_reactor C11_L04_reactor LABEL reference TIMEOUT 30)
  if(WIN32)
    add_executable(C11_L04_iocp iocp.cpp)
    c11_target(C11_L04_iocp)

if(C11_BUILD_REFERENCE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(C11_L04_epoll epoll.cpp)
  c11_target(C11_L04_epoll)
  c11_test(C11_L04_epoll C11_L04_epoll LABEL reference)
  option(C11_ENABLE_URING "Build Linux liburing socket branch" OFF)
  if(C11_ENABLE_URING)
    find_path(C11_URING_INCLUDE liburing.h REQUIRED)
    find_library(C11_URING_LIBRARY uring REQUIRED)
    add_executable(C11_L04_uring uring.cpp)
    c11_target(C11_L04_uring)
    target_include_directories(C11_L04_uring PRIVATE "${C11_URING_INCLUDE}")
    target_link_libraries(C11_L04_uring PRIVATE "${C11_URING_LIBRARY}")
    c11_test(C11_L04_uring C11_L04_uring LABEL reference ALLOW_SKIP)
  endif()
endif()
