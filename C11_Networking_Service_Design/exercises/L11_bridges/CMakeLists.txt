cmake_minimum_required(VERSION 3.28)
project(C11_L11_bridges LANGUAGES CXX)
include(../cmake/StudySetup.cmake)
set(C11_ASIO_SOURCE "" CACHE PATH "Standalone Asio1.38.2 checkout for unchanged C09 client")
if(C11_BUILD_REFERENCE AND C11_ASIO_SOURCE)
  c11_require_commit("${C11_ASIO_SOURCE}" 8806a6803cde7054c3049d3666d3ec36786568c5)
  set(legacy "${C11_REPO}/C09_Coroutines/exercises/Capstone4_rpc_framework")
  add_executable(C11_L11_legacy legacy.cpp "${legacy}/good/client.cpp" "${legacy}/good/protocol.cpp")
  c11_target(C11_L11_legacy)
  target_include_directories(C11_L11_legacy PRIVATE "${legacy}/include")
  target_include_directories(C11_L11_legacy SYSTEM PRIVATE "${C11_ASIO_SOURCE}/include")
  target_compile_definitions(C11_L11_legacy PRIVATE ASIO_STANDALONE)
  c11_test(C11_L11_legacy C11_L11_legacy LABEL reference)
endif()
if(C11_BUILD_REFERENCE AND C11_ENABLE_STDEXEC AND C11_ENABLE_BOOST)
  add_executable(C11_L11_sender sender.cpp)
  c11_target(C11_L11_sender)
  target_link_libraries(C11_L11_sender PRIVATE c11_boost c11_stdexec)
  if(MSVC)
    target_compile_options(C11_L11_sender PRIVATE /Zc:preprocessor)
  endif()
  c11_test(C11_L11_sender C11_L11_sender LABEL reference)
endif()
