cmake_minimum_required(VERSION 3.28)
project(c02_wiring_selfcheck LANGUAGES CXX)
option(C02_WIRING_BAD "Deliberately include an answer in the audit self-check" OFF)
if(C02_WIRING_BAD)
    add_executable(wiring_student src/student/bad.cpp)
else()
    add_executable(wiring_student src/student/main.cpp)
endif()
target_compile_features(wiring_student PRIVATE cxx_std_23)
set_target_properties(wiring_student PROPERTIES CXX_SCAN_FOR_MODULES OFF)
if(MSVC)
    target_compile_options(wiring_student PRIVATE /showIncludes /utf-8)
else()
    target_compile_options(wiring_student PRIVATE -H)
endif()
