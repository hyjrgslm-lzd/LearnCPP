cmake_minimum_required(VERSION 3.28)
project(L12_storage LANGUAGES CXX)
include(../cmake/StudySetup.cmake)

function(l12_configure target include_dir)
    add_executable(${target} checks/run_checks.cpp)
    target_include_directories(${target} PRIVATE checks ${include_dir})
    core_configure_target(${target})
endfunction()

if(CORE_STUDY_TEST_STUDENTS)
    l12_configure(L12_storage_student src/student)
    core_add_test(L12_storage_student COMMAND L12_storage_student LABELS student TIMEOUT 30)
else()
    add_executable(L12_storage_student EXCLUDE_FROM_ALL checks/run_checks.cpp)
    target_include_directories(L12_storage_student PRIVATE checks src/student)
    core_configure_target(L12_storage_student)
endif()

if(CORE_STUDY_BUILD_REFERENCE)
    l12_configure(L12_storage_reference src/reference)
    core_add_test(L12_storage_reference COMMAND L12_storage_reference LABELS reference TIMEOUT 30)

    add_executable(L12_storage_start_lifetime_as checks/start_lifetime_as_capability.cpp)
    core_configure_target(L12_storage_start_lifetime_as)
    core_add_test(L12_storage_start_lifetime_as
        COMMAND L12_storage_start_lifetime_as
        LABELS capability
        TIMEOUT 30)
    set_tests_properties(L12_storage_start_lifetime_as
        PROPERTIES SKIP_REGULAR_EXPRESSION "SKIP: __cpp_lib_start_lifetime_as not defined")

    l12_configure(L12_storage_validation_good validation/good)
    core_add_test(L12_storage_validation_good COMMAND L12_storage_validation_good LABELS validation TIMEOUT 30)

    l12_configure(L12_storage_validation_bad_noop validation/bad_noop)
    add_test(NAME L12_storage_validation_bad_noop_check
        COMMAND ${CMAKE_COMMAND}
            -DEXECUTABLE=$<TARGET_FILE:L12_storage_validation_bad_noop>
            -DEXPECTED_TEXT=slot must report engaged after successful construct
            -P ${CMAKE_CURRENT_LIST_DIR}/checks/expect_failure.cmake)
    set_tests_properties(L12_storage_validation_bad_noop_check PROPERTIES TIMEOUT 30 LABELS negative)
endif()
