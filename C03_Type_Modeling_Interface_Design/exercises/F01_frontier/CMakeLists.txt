cmake_minimum_required(VERSION 3.28)
project(F01_frontier LANGUAGES CXX)

include(CTest)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/StudySetup.cmake")

function(f01_capability target source)
    add_executable(${target} capabilities/${source})
    c03_configure_target(${target})
    if(MSVC)
        target_compile_options(${target} PRIVATE /std:c++latest)
    else()
        target_compile_options(${target} PRIVATE -std=c++2c)
    endif()
    c03_add_test(${target} COMMAND ${target} LABELS frontier TIMEOUT 30)
    set_tests_properties(${target} PROPERTIES SKIP_RETURN_CODE 77)
endfunction()
