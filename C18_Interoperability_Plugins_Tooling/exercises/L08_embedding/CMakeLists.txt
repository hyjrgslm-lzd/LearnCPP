cmake_minimum_required(VERSION 3.24)
project(C18_L08_embedding LANGUAGES C CXX)
include(../cmake/StudySetup.cmake)
include(../backends/PythonSupport.cmake)

if(C18_ENABLE_PYTHON)
  c18_python310_paths(c18_py_exe c18_py_include c18_py_libdir)
  add_library(c18_python_backend STATIC ../backends/python_backend.cpp)
  c18_target(c18_python_backend)
  target_include_directories(c18_python_backend PUBLIC "${C18_EXERCISES}/backends" PRIVATE "${c18_py_include}")
  get_filename_component(c18_py_home "${c18_py_libdir}/.." ABSOLUTE)
  target_compile_definitions(c18_python_backend PRIVATE C18_PYTHON_HOME="${c18_py_home}")
  if(WIN32)
    target_link_libraries(c18_python_backend PUBLIC "${c18_py_libdir}/python310.lib")
  endif()
endif()

function(c18_l08_variant variant implementation)
  if(NOT C18_ENABLE_PYTHON)
    return()
  endif()
  add_executable(c18_l08_${variant} checks.cpp)
  c18_target(c18_l08_${variant})
  target_include_directories(c18_l08_${variant} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/${implementation}" "${c18_py_include}")
  get_filename_component(c18_py_home "${c18_py_libdir}/.." ABSOLUTE)
  target_compile_definitions(c18_l08_${variant} PRIVATE C18_PYTHON_HOME="${c18_py_home}")
  if(WIN32)
    target_link_libraries(c18_l08_${variant} PRIVATE "${c18_py_libdir}/python310.lib")
  endif()
endfunction()

c18_l08_variant(student student)
if(C18_BUILD_REFERENCE)
  c18_l08_variant(reference reference)
  c18_l08_variant(good validation/good)
  c18_l08_variant(bad validation/bad)
endif()

if(C18_ENABLE_PYTHON AND C18_BUILD_REFERENCE)
  c18_add_test(c18_l08_reference c18_l08_reference LABELS "reference;python;embedding")
  c18_add_test(c18_l08_good c18_l08_good LABELS "good;python;embedding")
  c18_add_test(c18_l08_bad_rejected c18_l08_bad
    EXPECT_FAILURE "check failed: finalize was attempted before joining the Python worker" LABELS "checker;python;embedding")
endif()
if(C18_ENABLE_PYTHON AND C18_TEST_STUDENTS)
  c18_add_test(c18_l08_student c18_l08_student LABELS "student;python;embedding")
endif()
