cmake_minimum_required(VERSION 3.24)
project(C18_P1 LANGUAGES C CXX)
include(../cmake/StudySetup.cmake)
c18_require_native_plugin()
add_executable(c18_workbench main.cpp)
c18_target(c18_workbench)
target_link_libraries(c18_workbench PRIVATE ${CMAKE_DL_LIBS})
target_include_directories(c18_workbench PRIVATE "${C18_EXERCISES}/backends")
if(C18_ENABLE_PYTHON)
  if(NOT TARGET c18_python_backend)
    add_subdirectory(../L08_embedding "${CMAKE_CURRENT_BINARY_DIR}/python-embedding")
  endif()
  target_link_libraries(c18_workbench PRIVATE c18_python_backend)
  target_compile_definitions(c18_workbench PRIVATE C18_WITH_PYTHON)
endif()
if(C18_ENABLE_LUA)
  if(NOT TARGET c18_lua_backend)
    add_subdirectory(../L11_lua_stack "${CMAKE_CURRENT_BINARY_DIR}/lua-embedding")
  endif()
  target_link_libraries(c18_workbench PRIVATE c18_lua_backend)
  target_compile_definitions(c18_workbench PRIVATE C18_WITH_LUA)
endif()
add_dependencies(c18_workbench c18_native_plugin)
c18_add_command_test(c18_p1_native COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/checks.py" "$<TARGET_FILE:c18_workbench>" native LABELS "integration;native")
c18_add_command_test(c18_p1_plugin COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/checks.py" "$<TARGET_FILE:c18_workbench>" plugin "$<TARGET_FILE:c18_native_plugin>" LABELS "integration;plugin")
foreach(backend PYTHON LUA)
  if(C18_ENABLE_${backend})
    string(TOLOWER "${backend}" backend_lower)
    c18_add_command_test(c18_p1_${backend_lower} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/checks.py" "$<TARGET_FILE:c18_workbench>" "${backend_lower}" LABELS "integration;${backend_lower}")
  endif()
endforeach()
