cmake_minimum_required(VERSION 3.24)
project(C18_L03 LANGUAGES C CXX)
include(../cmake/StudySetup.cmake)
add_library(c18_l03_contract_plugin SHARED ../L04_plugin_shutdown/fixtures/good_plugin.cpp)
c18_target(c18_l03_contract_plugin)
target_compile_definitions(c18_l03_contract_plugin PRIVATE C18_PLUGIN_BUILD)
set(variants student)
if(C18_BUILD_REFERENCE)
  list(APPEND variants reference good bad)
endif()
foreach(variant IN LISTS variants)
  if(variant STREQUAL "good" OR variant STREQUAL "bad")
    set(source "validation/${variant}")
  else()
    set(source "${variant}")
  endif()
  add_executable(L03_${variant} checks.c implementation.cpp)
  c18_target(L03_${variant})
  target_include_directories(L03_${variant} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
  target_link_libraries(L03_${variant} PRIVATE c18_l03_contract_plugin)
  if(variant STREQUAL "student")
    if(C18_TEST_STUDENTS)
      c18_add_test(L03_student L03_student LABELS student)
    endif()
  elseif(variant STREQUAL "bad")
    c18_add_test(L03_bad_rejected L03_bad EXPECT_FAILURE "check failed: required function table" LABELS checker)
  else()
    c18_add_test(L03_${variant} L03_${variant} LABELS ${variant})
  endif()
endforeach()
