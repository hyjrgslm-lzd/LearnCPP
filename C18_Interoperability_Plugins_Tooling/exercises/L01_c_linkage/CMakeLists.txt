cmake_minimum_required(VERSION 3.24)
project(C18_L01 LANGUAGES C CXX)
include(../cmake/StudySetup.cmake)
set(variants student)
if(C18_BUILD_REFERENCE)
  list(APPEND variants reference good bad)
endif()
foreach(variant IN LISTS variants)
  if(variant STREQUAL "good" OR variant STREQUAL "bad")
    set(source "validation/${variant}")
  else()
    set(source "${variant}")
  endif()
  add_executable(L01_${variant} consumer.c implementation.cpp)
  c18_target(L01_${variant})
  target_include_directories(L01_${variant} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
  if(variant STREQUAL "student")
    if(C18_TEST_STUDENTS)
      c18_add_test(L01_student L01_student LABELS student)
    endif()
  elseif(variant STREQUAL "bad")
    c18_add_test(L01_bad_rejected L01_bad EXPECT_FAILURE "check failed: complete byte conversion" LABELS checker)
  else()
    c18_add_test(L01_${variant} L01_${variant} LABELS ${variant})
  endif()
endforeach()
