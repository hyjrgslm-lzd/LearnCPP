cmake_minimum_required(VERSION 3.24)
project(C18_F01 LANGUAGES C CXX)
include(../cmake/StudySetup.cmake)
option(C18_ENABLE_FREE_THREADED_CAPI "Build the CPython 3.13 free-threaded native subject in a separate build" OFF)
c18_add_command_test(c18_f01_subinterpreters COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_subject.py" subinterpreters ALLOW_SKIP LABELS "frontier;observation")
c18_add_command_test(c18_f01_free_capability COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/subject.py" free_threaded --probe ALLOW_SKIP LABELS "frontier;capability")
if(C18_ENABLE_FREE_THREADED_CAPI)
  if(CMAKE_VERSION VERSION_LESS 3.30 OR C18_ENABLE_PYTHON OR C18_ENABLE_LIMITED_API)
    message(FATAL_ERROR "Free-threaded subject needs CMake >=3.30 and a separate build with Full/Limited teaching lanes OFF")
  endif()
  set(Python3_FIND_ABI OFF ANY ANY ON)
  find_package(Python3 3.13...<3.14 REQUIRED COMPONENTS Interpreter Development.Module)
  execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import sysconfig; print(int(bool(sysconfig.get_config_var('Py_GIL_DISABLED'))))" OUTPUT_VARIABLE gil_disabled OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE probe_result)
  if(NOT probe_result EQUAL 0 OR NOT gil_disabled STREQUAL "1")
    message(FATAL_ERROR "The selected interpreter is not a CPython 3.13 free-threaded build")
  endif()
  Python3_add_library(_c18_free MODULE WITH_SOABI free_native.cpp)
  c18_target(_c18_free)
  target_compile_definitions(_c18_free PRIVATE Py_GIL_DISABLED=1)
  c18_add_command_test(c18_f01_free_subject COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_subject.py" free_threaded --module-dir "$<TARGET_FILE_DIR:_c18_free>" LABELS "frontier;native-subject")
endif()
