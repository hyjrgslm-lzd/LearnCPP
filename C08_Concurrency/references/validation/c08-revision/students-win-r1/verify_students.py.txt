void check_matrix_args(cs::numeric::input a, cs::numeric::input b, cs::numeric::output c, std::size_t n) {
    cs::check(a.size() == n * n && b.size() == n * n && c.size() == n * n, "matrix shape");
    cs::check(!cs::numeric::overlaps(a, c) && !cs::numeric::overlaps(b, c), "matrix output overlap");
}
void gemm_rows(cs::numeric::input a, cs::numeric::input b, cs::numeric::output c,
               std::size_t n, std::size_t block, std::size_t row_begin, std::size_t row_end) {
    for (std::size_t ii = row_begin; ii < row_end; ii += block) {
        const std::size_t i_end = std::min(row_end, ii + block);
        for (std::size_t kk = 0; kk < n; kk += block) {
            const std::size_t k_end = std::min(n, kk + block);
            for (std::size_t jj = 0; jj < n; jj += block) {
                const std::size_t j_end = std::min(n, jj + block);
                for (std::size_t i = ii; i < i_end; ++i)
                    for (std::size_t k = kk; k < k_end; ++k) {
                        const float aik = a[i * n + k];
                        for (std::size_t j = jj; j < j_end; ++j) c[i * n + j] += aik * b[k * n + j];
                    }
            }
        }
    }
}
void student_gemm_naive(cs::numeric::input a, cs::numeric::input b, cs::numeric::output c, std::size_t n) {
    check_matrix_args(a, b, c, n);
    for (std::size_t i = 0; i < n; ++i)
        for (std::size_t j = 0; j < n; ++j) {
            float sum = 0;
            for (std::size_t k = 0; k < n; ++k) sum += a[i * n + k] * b[k * n + j];
            c[i * n + j] = sum;
        }
}

}
void student_gemm_threaded(cs::numeric::input a, cs::numeric::input b, cs::numeric::output c,
                           std::size_t n, std::size_t block) {
    check_matrix_args(a, b, c, n);
    cs::check(block > 0, "block size");
    std::fill(c.begin(), c.end(), 0.0f);
    std::vector<std::future<void>> workers;
    const std::size_t threads = std::min<std::size_t>(2, n == 0 ? 1 : n);
    for (std::size_t t = 0; t < threads; ++t) {
        const std::size_t first = t * n / threads;
        const std::size_t last = (t + 1) * n / threads;
        workers.push_back(std::async(std::launch::async, [=] { gemm_rows(a, b, c, n, block, first, last); }));
    }
    for (auto& worker : workers) worker.get();
}

    else if (version == "par") student_gemm_par(a,b,c,n);
    else if (version == "sse2") throw std::runtime_error("optional SIMD not implemented by validation good overlay");
    else throw std::invalid_argument("unknown GEMM variant");
}
double student_sum(cs::numeric::input values, std::string_view version) {
    if (version == "threaded") {
        const auto mid = values.begin() + static_cast<std::ptrdiff_t>(values.size() / 2);
        auto left = std::async(std::launch::async, [=] { return std::accumulate(values.begin(), mid, 0.0); });
        const double right = std::accumulate(mid, values.end(), 0.0);
        return left.get() + right;
    }
    if (version == "par") {
#if CS_HAS_PARALLEL_ALGORITHMS
        return std::transform_reduce(std::execution::par, values.begin(), values.end(), 0.0, std::plus<>{},
                                     [](float value) { return static_cast<double>(value); });
#else
        throw std::runtime_error("parallel algorithms unavailable");
#endif
    }
    return std::transform_reduce(values.begin(), values.end(), 0.0, std::plus<>{},
                                 [](float value) { return static_cast<double>(value); });
}
void student_sort(std::span<int> values, bool parallel) {
    if (!parallel) { std::sort(values.begin(), values.end()); return; }
#if CS_HAS_PARALLEL_ALGORITHMS
    std::sort(std::execution::par, values.begin(), values.end());
#else
    throw std::runtime_error("parallel algorithms unavailable");
#endif
}

SANITIZER_MARKERS = (
    "AddressSanitizer",
    "ERROR: AddressSanitizer",
    "LeakSanitizer",
    "ThreadSanitizer",
    "UndefinedBehaviorSanitizer",
)


def has_sanitizer_report(text: str) -> bool:
    return any(marker in text for marker in SANITIZER_MARKERS)

def sha(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8")).hexdigest()


def apply_done_flags(text: str) -> str:
    return re.sub(r"(constexpr bool [A-Za-z0-9_]+ = )false;", r"\1true;", text)


def build_overlay(source: str, rel: str, mode: str) -> str:
    if mode == "initial":
        return source
    if rel in GOOD_FULL_FILES:
        text = GOOD_FULL_FILES[rel]
    else:
        text = apply_done_flags(source)
        replacements = GOOD_REPLACEMENTS.get(rel, {})
        for old, new in replacements.items():
            if old not in text:
                raise RuntimeError(f"{rel}: replacement anchor missing for good")
            text = text.replace(old, new)
    if mode == "bad":
        old, new, _ = BAD_MUTATIONS[rel]
        if old not in text:
            raise RuntimeError(f"{rel}: mutation anchor missing")
        text = text.replace(old, new)
    return text


def ensure_child(parent: Path, child: Path) -> None:
    resolved_parent = parent.resolve()
    resolved_child = child.resolve() if child.exists() else child.absolute().resolve().parent / child.name
    if resolved_child == resolved_parent or not resolved_child.is_relative_to(resolved_parent):
        raise RuntimeError(f"unsafe generated path: {resolved_child}")


def copy_tree(src: Path, dst: Path, sandbox: Path) -> None:
    ensure_child(sandbox, dst)
    if dst.resolve() == src.resolve():
        raise RuntimeError("refuse to copy source onto itself")
    if dst.exists():
        shutil.rmtree(dst)
    def ignore(_: str, names: list[str]) -> set[str]:
        return {name for name in names if name.startswith("build") or name == "__pycache__"}
    shutil.copytree(src, dst, ignore=ignore)


def keep_only_target(exercises: Path, target: str) -> None:
    keep = {"cmake", "include", "runtime_tests", target}
    for child in exercises.iterdir():
        if child.is_dir() and child.name not in keep:
            ensure_child(exercises, child)
            shutil.rmtree(child)


def executable_path(build: Path, target: str, config: str) -> Path:
    candidates = [
        build / config / f"{target}.exe",
        build / f"{target}.exe",
        build / config / target,
        build / target,
        build / target / config / f"{target}.exe",
        build / target / f"{target}.exe",
        build / target / config / target,
        build / target / target,
    ]
    for candidate in candidates:
        if candidate.exists():
            return candidate
    raise FileNotFoundError(f"cannot find executable for {target} under {build}")


def checked_process(command: list[str], timeout: int, output_dir: Path, stem: str) -> dict:
    result = run_process(command, timeout)
    output_dir.mkdir(parents=True, exist_ok=True)
    (output_dir / f"{stem}.stdout.txt").write_text(result["stdout"], encoding="utf-8")
    (output_dir / f"{stem}.stderr.txt").write_text(result["stderr"], encoding="utf-8")
    (output_dir / f"{stem}.json").write_text(json.dumps(result, ensure_ascii=False, indent=2), encoding="utf-8")
    return result


def configure_and_build(source: Path, build: Path, target: str, config: str, timeout: int, output_dir: Path) -> None:
    build.mkdir(parents=True, exist_ok=True)
    if not source.is_dir():
        raise NotADirectoryError(f"source directory does not exist: {source}")
    cmake_exe = shutil.which("cmake")
    if not cmake_exe:
        raise FileNotFoundError("cmake not found on PATH")
    generator = os.environ.get("C08_STUDENT_VALIDATION_GENERATOR")
    include_flag = os.environ.get("C08_STUDENT_VALIDATION_INCLUDE_FLAG", "/showIncludes" if os.name == "nt" else "-H")
    cmake = [cmake_exe, "-S", str(source), "-B", str(build), "-DBUILD_TESTING=ON",
             "-DCMAKE_EXPORT_COMPILE_COMMANDS=ON",
             f"-DCMAKE_CXX_FLAGS={include_flag}",
             "-DCONCURRENCY_STUDY_BUILD_REFERENCE=OFF",
             "-DCONCURRENCY_STUDY_TEST_STARTERS=ON",
             "-DCONCURRENCY_STUDY_FETCH_DEPS=OFF",
             "-DCONCURRENCY_STUDY_BUILD_BENCHMARKS=OFF"]
    if generator:
        cmake += ["-G", generator]
    result = checked_process(cmake, timeout, output_dir, "configure")
    if result["exit_code"] != 0 or result["timeout"] or result["cleanup_error"]:
        raise RuntimeError(result["stdout"] + result["stderr"] + result["error"])
    result = checked_process([cmake_exe, "--build", str(build), "--config", config, "--target", target],
                             timeout, output_dir, "build")
    if result["exit_code"] != 0 or result["timeout"] or result["cleanup_error"]:
        raise RuntimeError(result["stdout"] + result["stderr"] + result["error"])
    return result


def collect_dependency_trace(build: Path, compiler_output: str = "") -> list[str]:
    roots: set[str] = set()
    normalized_output = compiler_output.replace("\\", "/")
    for match in re.finditer(r"[A-Za-z]:/[^\r\n:]*?\.(?:h|hpp|hh|hxx|cpp)\b", normalized_output, re.IGNORECASE):
        roots.add(match.group(0).strip(' "'))
    for pattern in ("*.d", "*.obj.d", "*.tlog"):
        for depfile in build.rglob(pattern):
            try:
                text = depfile.read_text(encoding="utf-8", errors="ignore")
            except OSError:
                continue
            normalized = text.replace("\\", "/")
            for token in re.split(r"[\s;]+", normalized):
                item = token.strip(' \"')
                if item.lower().endswith(('.h', '.hpp', '.hh', '.hxx', '.cpp')):
                    roots.add(item)
    return sorted(roots)


def assert_student_dependency_trace(target: str, build: Path, output_dir: Path, build_result: dict) -> list[str]:
    deps = collect_dependency_trace(build, build_result.get("stdout", "") + build_result.get("stderr", ""))
    (output_dir / "dependencies.txt").write_text("\n".join(deps) + ("\n" if deps else ""), encoding="utf-8")
    if not deps:
        raise RuntimeError(f"{target}: no compiler dependency trace found for Student include audit")
    banned = ("/reference.hpp", "/solution.cpp", "/validation/")
    leaked = [dep for dep in deps if any(bad in dep.lower() for bad in banned)]
    if leaked:
        raise RuntimeError(f"{target}: forbidden Student dependency: {leaked}")
    return deps

def verify_one(course: Path, build_root: Path, target: str, mode: str, config: str, timeout: int) -> dict[str, object]:
    case_root = build_root / mode
    work = case_root / "course"
    build = case_root / "build"
    output_dir = case_root / "logs"
    copy_tree(course, work, case_root)
    ensure_child(case_root, build)
    if build.exists():
        shutil.rmtree(build)
    source = work / "exercises"
    keep_only_target(source, target)
    source_digests: dict[str, str] = {}
    digests: dict[str, str] = {}
    include_trace: dict[str, list[str]] = {}
    overlay_rel: str | None = None
    for rel in {**GOOD_REPLACEMENTS, **GOOD_FULL_FILES, **BAD_MUTATIONS}:
        exercise = Path(rel).parts[0]
        if exercise != target:
            continue
        overlay_rel = rel
        path = source / rel
        original = path.read_text(encoding="utf-8")
        source_digests[rel] = sha(original)
        overlay = build_overlay(original, rel, mode)
        path.write_text(overlay, encoding="utf-8", newline="\n")
        if mode != "initial":
            digests[rel] = sha(overlay)
            include_trace[rel] = [line.strip() for line in overlay.splitlines() if line.lstrip().startswith("#include")]
    if mode in {"good", "bad"} and not digests:
        rel = f"{target}/main.cpp"
        path = source / rel
        original = path.read_text(encoding="utf-8")
        source_digests[rel] = sha(original)
        overlay = build_overlay(original, rel, mode)
        path.write_text(overlay, encoding="utf-8", newline="\n")
        digests[rel] = sha(overlay)
        include_trace[rel] = [line.strip() for line in overlay.splitlines() if line.lstrip().startswith("#include")]
        overlay_rel = rel
    if mode == "good":
        banned = ("reference.hpp", "solution.cpp")
        for rel, includes in include_trace.items():
            if any(any(bad in line for bad in banned) for line in includes):
                raise RuntimeError(f"{target}: good overlay includes Reference path: {includes}")
    build_result = configure_and_build(source / target, build, target, config, timeout, output_dir)
    exe = executable_path(build, target, config)
    result = checked_process([str(exe)], timeout, output_dir, "run")
    combined = result["stdout"] + result["stderr"]
    dependency_trace: list[str] = []
    clean_failure = not result["timeout"] and not result["cleanup_error"] and not has_sanitizer_report(combined)
    if mode == "good":
        dependency_trace = assert_student_dependency_trace(target, build, output_dir, build_result)
        passed = result["exit_code"] == 0 and clean_failure
    elif mode == "bad":
        if overlay_rel is None:
            raise RuntimeError(f"{target}: no bad overlay selected")
        expected = BAD_MUTATIONS[overlay_rel][2]
        passed = result["exit_code"] == 1 and expected in combined and clean_failure
    else:
        expected = INITIAL_EXPECTED[target]
        passed = result["exit_code"] == 1 and expected in combined and clean_failure
    if not passed:
        raise RuntimeError(f"{target} {mode}: unexpected result {json.dumps(result, ensure_ascii=False)}")
    return {
        "target": target,
        "mode": mode,
        "exit": result["exit_code"],
        "exe": str(exe),
        "source_sha256": source_digests,
        "overlay_sha256": digests,
        "include_trace": include_trace,
        "dependency_trace_count": len(dependency_trace),
        "dependency_trace_sha256": sha("\n".join(dependency_trace)),
        "student_dependency_trace": [dep for dep in dependency_trace if "/course/exercises/" in dep.lower()],
        "logs": str(output_dir),
        "stdout_sha256": sha(result["stdout"]),
        "stderr_sha256": sha(result["stderr"]),
    }


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--course", type=Path, default=Path(__file__).resolve().parents[2])
    parser.add_argument("--build-root", type=Path, default=Path(__file__).resolve().parents[2] / "build" / "student-validation")
    parser.add_argument("--target", action="append", choices=EXERCISES)
    parser.add_argument("--mode", action="append", choices=["initial", "good", "bad"])
    parser.add_argument("--config", default="Release")
    parser.add_argument("--timeout", type=int, default=30)
    args = parser.parse_args()
    course = args.course.resolve()
    build_root = args.build_root.resolve()
    if build_root.exists() and any(build_root.iterdir()):
        stamp = dt.datetime.now().strftime("run-%Y%m%d-%H%M%S-%f")
        build_root = build_root / stamp
        while build_root.exists():
            stamp = dt.datetime.now().strftime("run-%Y%m%d-%H%M%S-%f")
            build_root = args.build_root.resolve() / stamp
    exercises_dir = course / "exercises"
    expected_rel = {
        f"{name}/student.hpp" if (exercises_dir / name / "student.hpp").exists() else f"{name}/main.cpp"
        for name in EXERCISES
    }
    overlay_rel = set(GOOD_REPLACEMENTS) | set(GOOD_FULL_FILES)
    bad_rel = set(BAD_MUTATIONS)
    missing_initial = sorted(set(EXERCISES) - set(INITIAL_EXPECTED))
    missing_good = sorted(expected_rel - overlay_rel)
    missing_bad = sorted(expected_rel - bad_rel)
    if missing_good or missing_bad or missing_initial:
        raise RuntimeError(f"validation coverage gap: good={missing_good}, bad={missing_bad}, initial={missing_initial}")
    targets = args.target or EXERCISES
    modes = args.mode or ["initial", "good", "bad"]
    report = []
    for target in targets:
        for mode in modes:
            item = verify_one(course, build_root / target, target, mode, args.config, args.timeout)
            report.append(item)
            print(json.dumps(item, ensure_ascii=False), flush=True)

    (build_root / "summary.json").write_text(json.dumps(summary, ensure_ascii=False, indent=2), encoding="utf-8")
