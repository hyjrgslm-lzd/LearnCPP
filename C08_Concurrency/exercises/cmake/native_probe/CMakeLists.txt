cmake_minimum_required(VERSION 3.28)
project(cs_native_probe LANGUAGES CXX)
unset(CMAKE_CXX_STANDARD)
add_executable(cs_native_probe ../feature_probes.cpp)
target_compile_definitions(cs_native_probe PRIVATE ${CS_FEATURE}=1)
if(MSVC)
    target_compile_options(cs_native_probe PRIVATE /std:c++latest /utf-8 /EHsc /Zc:__cplusplus /Zc:preprocessor)
else()
    target_compile_options(cs_native_probe PRIVATE -std=c++2c)
endif()
